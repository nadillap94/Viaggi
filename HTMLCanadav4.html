 document.getElementById('ganttProgressBar').style.width = pct + '%';
 document.getElementById('ganttPctText').textContent = pct + '%';

}

function registerVisitedDay(dayNum) {
 const visited = JSON.parse(localStorage.getItem('canada_visited_days') || '[]');
 if (!visited.includes(dayNum)) {
   visited.push(dayNum);
   localStorage.setItem('canada_visited_days', JSON.stringify(visited));
 }
 updatePassportUI();
}

function updatePassportUI() {
 const visited = JSON.parse(localStorage.getItem('canada_visited_days') || '[]');
 for (let i = 1; i <= 12; i++) {
   const stamp = document.getElementById(`stamp-day-${i}`);
   if (stamp) {
     const isDone = visited.includes(i);
     stamp.classList.toggle('bg-glacier', isDone);
     stamp.classList.toggle('text-white', isDone);
     stamp.classList.toggle('bg-frost', !isDone);
     stamp.classList.toggle('text-glacier/50', !isDone);
   }
 }
}

function initSnowfall() {
 const container = document.getElementById('snowfall');
 for (let i = 0; i < 22; i++) {
   const f = document.createElement('div');
   f.className = 'flake';
   f.textContent = '❄';
   const size = 10 + Math.random() * 12;
   f.style.left = Math.random() * 100 + 'vw';
   f.style.fontSize = size + 'px';
   f.style.opacity = (0.25 + Math.random() * 0.4).toString();
   const duration = 12 + Math.random() * 14;
   const delay = Math.random() * -20;
   f.animate([
     { transform: 'translateY(-10px) translateX(0)' },
     { transform: `translateY(105vh) translateX(${(Math.random() * 80 - 40)}px)` }
   ], {
     duration: duration * 1000,
     delay: delay * 1000,
     iterations: Infinity,
     easing: 'linear'
   });
   container.appendChild(f);
 }
}
