       function loadNotes() {
           const notes = JSON.parse(localStorage.getItem('expedition_notes') || '[]');
           const list = document.getElementById('notesList');
           if (notes.length === 0) {
               list.innerHTML = `<li class="text-slate-500 italic p-2 border border-dashed border-arctic-800">Nessuna nota personale registrata. Aggiungi promemoria qui sopra.</li>`;
               return;
           }
           list.innerHTML = notes.map((n, i) => `
               <li class="flex items-center justify-between p-2.5 bg-arctic-900 border border-arctic-800">
                   <span class="text-slate-200">${n}</span>
                   <button onclick="removeNote(${i})" class="text-red-400 hover:text-red-300 font-bold ml-4"><i class="fa-solid fa-trash-can"></i></button>
               </li>
           `).join('');

       function addPersonalNote() {
           const input = document.getElementById('noteInput');
           const val = input.value.trim();
           if (!val) return;
           const notes = JSON.parse(localStorage.getItem('expedition_notes') || '[]');
           notes.push(val);
           localStorage.setItem('expedition_notes', JSON.stringify(notes));
           input.value = '';
           loadNotes();
       }

       function removeNote(index) {
           const notes = JSON.parse(localStorage.getItem('expedition_notes') || '[]');
           notes.splice(index, 1);
           localStorage.setItem('expedition_notes', JSON.stringify(notes));
           loadNotes();
       }

       function toggleTaskState(taskId) {
           const savedState = JSON.parse(localStorage.getItem('gantt_task_state') || '{}');
           savedState[taskId] = !savedState[taskId];
           localStorage.setItem('gantt_task_state', JSON.stringify(savedState));
           renderGanttTasks();
       }

       function updateGanttProgress() {
           const savedState = JSON.parse(localStorage.getItem('gantt_task_state') || '{}');
           const total = ganttTasks.length;
           const completed = Object.values(savedState).filter(Boolean).length;
           const pct = Math.round((completed / total) * 100);

           document.getElementById('ganttProgressBar').style.width = pct + '%';
           document.getElementById('ganttProgressText').textContent = `${completed} / ${total} Task Completati (${pct}%)`;
       }
